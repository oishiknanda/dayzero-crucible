+    const ms = v > 1e12 ? v : v > 1e9 ? v * 1000 : NaN;
+    return Number.isFinite(ms) ? ms : null;
+  }
+  if (typeof v !== "string") return null;
+  const t = v.toLowerCase();
+  if (t.includes("today")) return now;
+  if (t.includes("yesterday")) return now - 86400000;
+  const ago = /(\d+)\s*\+?\s*(day|week|hour)s?\s+ago/.exec(t);
+  if (ago) return now - parseInt(ago[1], 10) * (ago[2] === "week" ? 7 : ago[2] === "hour" ? 1 / 24 : 1) * 86400000;

+  return Number.isFinite(ms) ? ms : null;

+/** Finds arrays of job-like objects anywhere inside a JSON response. */
+export function jobsFromJson(json, base, now = Date.now()) {
+  const out = [];
+  const seen = new Set();
+  (function walk(node, depth) {
+    if (depth > 9 || node == null || typeof node !== "object") return;
+    if (Array.isArray(node)) {
+      const objs = node.filter(isObj);
+      if (objs.length >= 2) {
+        const jobs = objs.map((o) => toJob(o, base, now)).filter(Boolean);
+        if (jobs.length >= 2 && jobs.length >= objs.length * 0.6) {
+          for (const j of jobs) if (!seen.has(j.url)) { seen.add(j.url); out.push(j); }
+          return;
+        }
+      }
+      for (const v of node) walk(v, depth + 1);
+      return;
+    }
+    for (const k of Object.keys(node)) walk(node[k], depth + 1);
+  })(json, 0);
+  return out;

+/** schema.org JobPosting markup (script type="application/ld+json"). */
+export function jobsFromJsonLd(blocks, base, now = Date.now()) {
+  const out = [];
+  const visit = (n) => {
+    if (Array.isArray(n)) return n.forEach(visit);
+    if (!isObj(n)) return;
+    if (n["@graph"]) visit(n["@graph"]);
+    const type = n["@type"];
+    if (type === "JobPosting" || (Array.isArray(type) && type.includes("JobPosting"))) {
+      const url = absolute(typeof n.url === "string" ? n.url : typeof n.sameAs === "string" ? n.sameAs : base, base);
+      const title = text(n.title);
+      if (title && url) out.push({ title, url, location: text(n.jobLocation), publishedMs: parseDate(n.datePosted, now) });
+    }
+  };
+  for (const b of blocks) { try { visit(JSON.parse(b)); } catch { /* ignore bad markup */ } }
+  return out;
+}

+/** Last resort: links that look like individual job pages. Undated. */
+export function jobsFromLinks(links, base) {
+  const out = [];
+  const seen = new Set();
+  for (const l of links) {
+    const url = absolute(l.href, base);
+    const title = (l.text || "").replace(/\s+/g, " ").trim();
+    if (!url || seen.has(url) || title.length < 6 || title.length > 120) continue;
+    if (!/\/(job|jobs|position|positions|requisition|opening|openings|vacancy)\b|\/careers\/.+\d/i.test(new URL(url).pathname)) continue;
+    seen.add(url);
+    out.push({ title, url, location: "", publishedMs: null });
+  }
+  return out;
+}

+export type CrawledResult = { jobs: Alert[]; companies: Record<string, number> };

+const STALE_AFTER_MS = 3 * DAY;
+
+export function crawledUrl(): string {
+  return process.env.CRAWLED_JOBS_URL ||
+    "https://raw.githubusercontent.com/oishiknanda/dayzero-crucible/crawl-data/crawled-jobs.json";

+export function parseCrawled(file: unknown, names: string[], now: number): CrawledResult {
+  const empty: CrawledResult = { jobs: [], companies: {} };
+  if (!file || typeof file !== "object") return empty;
+  const f = file as { generatedAt?: unknown; companies?: Record<string, { ok?: boolean }>; jobs?: unknown };
+  const generated = typeof f.generatedAt === "string" ? Date.parse(f.generatedAt) : NaN;
+  if (!Number.isFinite(generated) || now - generated > STALE_AFTER_MS) return empty;
+  const jobs: Alert[] = [];
+  const companies: Record<string, number> = {};
+  for (const raw of Array.isArray(f.jobs) ? f.jobs : []) {
+    const j = raw as Record<string, unknown>;
+    const company = typeof j.company === "string" ? j.company : "";
+    const published = typeof j.publishedAt === "string" ? Date.parse(j.publishedAt) : NaN;
+    const url = typeof j.jobUrl === "string" ? j.jobUrl : "";
+    if (names.indexOf(company) === -1 || typeof j.title !== "string" || typeof j.id !== "string") continue;
+    if (!Number.isFinite(published) || published < now - MAX_AGE_MS || published > now + DAY) continue;
+    if (!/^https:\/\//i.test(url)) continue;
+    jobs.push({
+      id: j.id, company, title: j.title, department: null,
+      location: typeof j.location === "string" && j.location ? j.location : "India",
+      employmentType: null, workplaceType: null,
+      publishedAt: new Date(published).toISOString(), jobUrl: url, viaAgent: true,
+    });
+    companies[company] = (companies[company] || 0) + 1;
+  }
+  return { jobs, companies };
+}

+export async function fetchCrawled(names: string[], now: number): Promise<CrawledResult> {
+  const headers: Record<string, string> = { Accept: "application/json" };

+  const res = await fetch(crawledUrl(), { headers, signal: AbortSignal.timeout(10000), next: { revalidate: 300 } } as RequestInit);
+  if (!res.ok) throw new Error(`crawled jobs returned ${res.status}`);
+  return parseCrawled(await res.json(), names, now);
